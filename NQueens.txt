
import java.util.Scanner;

public class NQueens {

    static int n;
    static int board[];

    // Check whether queen can be placed safely
    static boolean isSafe(int row, int col) {

        for (int i = 0; i < row; i++) {

            // Check same column
            if (board[i] == col) {
                return false;
            }

            // Check diagonals
            if (Math.abs(board[i] - col) == Math.abs(i - row)) {
                return false;
            }
        }

        return true;
    }

    // Solve N-Queens using Backtracking
    static boolean solve(int row) {

        if (row == n) {
            return true;
        }

        for (int col = 0; col < n; col++) {

            if (isSafe(row, col)) {

                board[row] = col;

                if (solve(row + 1)) {
                    return true;
                }

                // Backtracking
                board[row] = -1;
            }
        }

        return false;
    }

    // Display chessboard
    static void display() {

        System.out.println("Solution:");

        for (int i = 0; i < n; i++) {

            for (int j = 0; j < n; j++) {

                if (board[i] == j) {
                    System.out.print(" Q ");
                } else {
                    System.out.print(" . ");
                }
            }

            System.out.println();
        }
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of queens: ");
        n = sc.nextInt();

        if (n < 1) {
            System.out.println("Enter a positive number.");
            sc.close();
            return;
        }

        board = new int[n];

        for (int i = 0; i < n; i++) {
            board[i] = -1;
        }

        if (solve(0)) {
            display();
        } else {
            System.out.println("No solution exists.");
        }

        sc.close();
    }
}
